    let pdfWorkerBlobUrl = null;
    async function configurePdfJsWorker() {
      if (pdfWorkerBlobUrl) return;
      try {
        const workerUrl = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
        const response = await fetch(workerUrl);
        const scriptText = await response.text();
        const blob = new Blob([scriptText], { type: 'application/javascript' });
        pdfWorkerBlobUrl = URL.createObjectURL(blob);
        pdfjsLib.GlobalWorkerOptions.workerSrc = pdfWorkerBlobUrl;
      } catch (err) {
        pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
      }
    }

    async function handlePdfFile(file) {
      if (!file) return;
      const fileName = file.name || '';
      const isPdf = fileName.toLowerCase().endsWith('.pdf') || (file.type && file.type.includes('pdf'));
      const isText = fileName.toLowerCase().endsWith('.txt') || (file.type && file.type.includes('text/plain'));

      if (isText) {
        const text = await file.text();
        appState.extractedRawText = text.trim();
        extractedTextArea.value = appState.extractedRawText;
        appState.lessonTitle = fileName.replace(/\.[^/.]+$/, "");
        el('examTopicName').textContent = `موضوع الدرس: ${appState.lessonTitle}`;
        updateCharCount();
        showToast('تم تحميل النص بنجاح!', 'success');
        pdfFileInput.value = '';
        return;
      }

      if (!isPdf) {
        showToast('يرجى اختيار ملف بصيغة PDF صالحة', 'error');
        pdfFileInput.value = '';
        return;
      }

      uploadIdleState.classList.add('hidden');
      uploadProcessingState.classList.remove('hidden');
      uploadProgressBar.style.width = '20%';
      uploadStatusText.textContent = `جارٍ فتح الملف: ${fileName.slice(0, 30)}...`;

      try {
        await configurePdfJsWorker();
        const fileData = await file.arrayBuffer();
        uploadProgressBar.style.width = '45%';
        uploadStatusText.textContent = 'جارٍ قراءة النصوص والصفحات...';

        const pdf = await pdfjsLib.getDocument({
          data: new Uint8Array(fileData),
          cMapUrl: 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/cmaps/',
          cMapPacked: true
        }).promise;

        let fullText = '';
        const totalPages = pdf.numPages;

        for (let i = 1; i <= totalPages; i++) {
          const page = await pdf.getPage(i);
          const textContent = await page.getTextContent();
          const pageText = textContent.items.map(item => item.str).join(' ');
          fullText += pageText + '\n\n';
          const progressPercent = Math.min(95, Math.round((i / totalPages) * 100));
          uploadProgressBar.style.width = `${progressPercent}%`;
          uploadStatusText.textContent = `جارٍ استخراج صفحة ${i} من ${totalPages}...`;
        }

        uploadProgressBar.style.width = '100%';

        if (fullText.trim().length < 15) {
          showToast('هذا الملف عبارة عن صور ممسوحة ضوئياً (Scanned). يرجى نسخ النص ولصقه بالمربع.', 'error');
        } else {
          appState.extractedRawText = fullText.trim();
          extractedTextArea.value = appState.extractedRawText;
          appState.lessonTitle = fileName.replace(/\.[^/.]+$/, "");
          el('examTopicName').textContent = `موضوع الدرس: ${appState.lessonTitle}`;
          updateCharCount();
          showToast(`تم استخراج نصوص ${totalPages} صفحة بنجاح!`, 'success');
        }
      } catch (err) {
        console.error('PDF Error:', err);
        showToast('تعذر استخراج النصوص تلقائياً؛ يمكنك لصق النص مباشرة في المربع', 'error');
      } finally {
        uploadProcessingState.classList.add('hidden');
        uploadIdleState.classList.remove('hidden');
        pdfFileInput.value = '';
      }
    }

    // Heuristic Bloom Question Synthesizer
    function synthesizePedagogicalQuestions(rawText, batchNum = 1) {
      const questions = [];
      const sentences = rawText.split(/[.\n؛!؟]/).map(s => s.trim()).filter(s => s.length > 25);
      if (sentences.length === 0) return [];

      const config = {
        truefalse: { enabled: el('type-truefalse').checked, count: parseInt(el('count-truefalse').value) || 2 },
        complete: { enabled: el('type-complete').checked, count: parseInt(el('count-complete').value) || 2 },
        explain: { enabled: el('type-explain').checked, count: parseInt(el('count-explain').value) || 2 },
        results: { enabled: el('type-results').checked, count: parseInt(el('count-results').value) || 2 },
        prove: { enabled: el('type-prove').checked, count: parseInt(el('count-prove').value) || 2 },
        terms: { enabled: el('type-terms').checked, count: parseInt(el('count-terms').value) || 2 },
        definitions: { enabled: el('type-definitions').checked, count: parseInt(el('count-definitions').value) || 2 },
      };

      let idCounter = 1;

      // Type 1: صح أم خطأ
      if (config.truefalse.enabled) {
        for (let i = 0; i < config.truefalse.count; i++) {
          const sent = sentences[(i * 3 + batchNum) % sentences.length] || sentences[0];
          const isTrue = (i % 2 === 0);
          let statement = sent;
          let correction = "العبارة صحيحة تماماً وتتسق مع المنطق العلمي وسياق النص.";
          let rationale = "التحليل: يؤكد النص صحة الفرضية؛ حيث ارتبطت الأسباب بالمسببات بصورة متطابقة.";

          if (!isTrue) {
            statement = sent.replace(/(أدى إلى|ترتب على|انطلقت|أحدثت|ساهمت|ظهرت)/, "لم تؤدِ إطلاقاً إلى").replace(/(وفرة|زيادة|انخفاض)/, "تراجع وانعدام");
            correction = "التصويب: العبارة خاطئة؛ فالصواب أن الواقع أثبت عكس ذلك كما ورد في النص الأصلي.";
            rationale = `التعليل التربوي: الخلل في العبارة يكمن في نفي النتيجة؛ حيث أثبت الدرس (${sent}).`;
          }

          questions.push({
            id: idCounter++,
            typeKey: 'truefalse',
            typeName: 'ضع علامة (صح) أو (خطأ) مع تصويب الخطأ وعلل إجابتك',
            bloom: 'evaluate',
            bloomText: bloomTaxonomy.evaluate.label,
            bloomColor: bloomTaxonomy.evaluate.color,
            bloomIcon: bloomTaxonomy.evaluate.icon,
            prompt: `«${statement}»`,
            expectedAnswer: isTrue ? "العبارة: (صحيحة ✓)" : "العبارة: (خاطئة ✗)",
            correctionNote: isTrue ? "لا يوجد تصويب (العبارة دقيقة)." : correction,
            rationale: rationale,
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 2: أكمل
      if (config.complete.enabled) {
        for (let i = 0; i < config.complete.count; i++) {
          const sent = sentences[(i * 2 + 1 + batchNum) % sentences.length] || sentences[1 % sentences.length];
          const words = sent.split(' ').filter(w => w.length > 3);
          const targetWord = words[Math.floor(words.length / 2)] || "العوامل الجوهرية";
          const maskedPrompt = sent.replace(targetWord, ' (....................) ');

          questions.push({
            id: idCounter++,
            typeKey: 'complete',
            typeName: 'أكمل العبارات الآتية بما يناسبها (فهم وتكامل تركيبي)',
            bloom: 'understand',
            bloomText: bloomTaxonomy.understand.label,
            bloomColor: bloomTaxonomy.understand.color,
            bloomIcon: bloomTaxonomy.understand.icon,
            prompt: maskedPrompt,
            expectedAnswer: `المصطلح أو المفهوم المكمل: [ ${targetWord} ]`,
            correctionNote: `يقبل أي مصطلح مرادف يؤدي إلى المعنى التكاملي التالي: «${sent}».`,
            rationale: 'التعليل التربوي: يهدف السؤال لاختبار الربط السياقي وتكامل الفكرة دون الاعتماد على الحفظ الآلي.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 3: بم تفسر
      if (config.explain.enabled) {
        for (let i = 0; i < config.explain.count; i++) {
          const sent = sentences[(i * 4 + batchNum) % sentences.length] || sentences[0];
          questions.push({
            id: idCounter++,
            typeKey: 'explain',
            typeName: 'بم تفسر / علل (استهداف مهارات التحليل والاستنتاج)',
            bloom: 'analyze',
            bloomText: bloomTaxonomy.analyze.label,
            bloomColor: bloomTaxonomy.analyze.color,
            bloomIcon: bloomTaxonomy.analyze.icon,
            prompt: `علل: ما هي الأسباب والعوامل الجوهرية التي أدت إلى: «${sent.slice(0, 100)}...»؟`,
            expectedAnswer: `التحليل السببي: يعود ذلك إلى التفاعل المباشر بين العوامل الموضحة في النص والتي خلقت بيئة مواتية لتشكل هذه الظاهرة.`,
            correctionNote: 'المحكات: ذكر السبب المباشر وتوضيح الظروف المحفزة.',
            rationale: 'يقيس السؤال قدرة المتعلم على تفكيك سلاسل العلة والمعلول بدقة استنتاجية.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 4: ما النتائج المترتبة
      if (config.results.enabled) {
        for (let i = 0; i < config.results.count; i++) {
          const sent = sentences[(i * 3 + 2 + batchNum) % sentences.length] || sentences[0];
          questions.push({
            id: idCounter++,
            typeKey: 'results',
            typeName: 'ما النتائج المترتبة على...؟ (تفكير ناقد وربط الآثار)',
            bloom: 'analyze',
            bloomText: bloomTaxonomy.analyze.label,
            bloomColor: bloomTaxonomy.analyze.color,
            bloomIcon: bloomTaxonomy.analyze.icon,
            prompt: `ما النتائج المباشرة والآثار اللاحقة المترتبة على: «${sent.slice(0, 95)}...»؟`,
            expectedAnswer: `النتائج المستخلصة: أدى ذلك إلى تحولات هيكلية ملموسة، مع إحداث آثار تراكمية غيرت المسار المعتاد وفرضت واقعاً تنظيمياً جديداً.`,
            correctionNote: 'الربط بين الأثر اللحظي والمردود بعيد المدى.',
            rationale: 'يقيس مهارة التنبؤ والتقييم العقلاني لمآلات الأحداث.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 5: دلل
      if (config.prove.enabled) {
        for (let i = 0; i < config.prove.count; i++) {
          const sent = sentences[(i * 2 + batchNum) % sentences.length] || sentences[0];
          questions.push({
            id: idCounter++,
            typeKey: 'prove',
            typeName: 'دلل تاريخياً / جغرافياً / علمياً على صحة العبارة',
            bloom: 'apply',
            bloomText: bloomTaxonomy.apply.label,
            bloomColor: bloomTaxonomy.apply.color,
            bloomIcon: bloomTaxonomy.apply.icon,
            prompt: `دلل بالأدلة والشواهد الواقعية من محتوى الدرس على صحة: «${sent.slice(0, 110)}»`,
            expectedAnswer: `الدليل والشاهد: الدليل القاطع يكمن في الشواهد الميدانية والوقائع المسجلة التي أثبتت دقة هذه الفرضية وتطابقها التام مع النتائج.`,
            correctionNote: 'يقبل أي استدلال منطقي أو شواهد مذكورة بالنص.',
            rationale: 'يقيس الاستدلال بالبراهين قدرة الطالب على نقل الفكرة إلى حيز الإثبات العلمي.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 6: اكتب ما تشير إليه العبارة
      if (config.terms.enabled) {
        for (let i = 0; i < config.terms.count; i++) {
          const sent = sentences[(i * 5 + 1 + batchNum) % sentences.length] || sentences[0];
          questions.push({
            id: idCounter++,
            typeKey: 'terms',
            typeName: 'اكتب ما تشير إليه العبارات الآتية (مفاهيم مركبة وتطبيقات)',
            bloom: 'create',
            bloomText: bloomTaxonomy.create.label,
            bloomColor: bloomTaxonomy.create.color,
            bloomIcon: bloomTaxonomy.create.icon,
            prompt: `[عبارة دالة]: «تحول جوهري يعبر عن: ${sent.slice(0, 120)}» فما هو المفهوم المركب المعبر عن ذلك؟`,
            expectedAnswer: `المفهوم المقصود: [ المفهوم الجوهري المعبر عن القضية المطروحة في الدرس ]`,
            correctionNote: 'التركيز على الدلالة الاصطلاحية وليس مجرد التسمية الحرفية.',
            rationale: 'يقيس القدرة على التجريد الذهني وتكثيف سياق سردي كامل في مصطلح علمي محدد.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      // Type 7: ما المقصود بـ
      if (config.definitions.enabled) {
        for (let i = 0; i < config.definitions.count; i++) {
          const sent = sentences[(i * 3 + batchNum) % sentences.length] || sentences[0];
          const keywords = sent.split(' ').filter(w => w.length > 5);
          const concept = keywords[0] ? `«${keywords[0]} ${keywords[1] || ''}»` : 'المفهوم المحوري في الدرس';

          questions.push({
            id: idCounter++,
            typeKey: 'definitions',
            typeName: 'ما المقصود بـ...؟ (صياغة المفاهيم بأسلوب الفهم لا الحفظ)',
            bloom: 'understand',
            bloomText: bloomTaxonomy.understand.label,
            bloomColor: bloomTaxonomy.understand.color,
            bloomIcon: bloomTaxonomy.understand.icon,
            prompt: `ما المقصود بـ ${concept} في ضوء ما ورد بالدرس؟ اشرح بأسلوبك التحليلي الخاص مبيناً خصائصه الرئيسية.`,
            expectedAnswer: `المقصود التفسيري: هو حالة أو منظومة ديناميكية تعبر عن (${sent.slice(0, 130)}...)، وتتميز بالفاعلية والتأثير المتواصل.`,
            correctionNote: 'الدرجة الكاملة تُمنح للأسلوب الشخصي المستوعب للشروط الفنية.',
            rationale: 'صياغة المفاهيم باللغة الخاصة للمتعلم هو أعلى مؤشرات الاستيعاب المعرفي.',
            userAnswer: '',
            isRevealed: false
          });
        }
      }

      return questions;
    }

    window.initiateQuestionGeneration = () => {
      const text = appState.extractedRawText.trim();
      if (!text || text.length < 40) return showToast('يرجى رفع ملف PDF أو كتابة محتوى دراسي معتبر', 'error');

      showToast('جارٍ تشغيل محرك بلوم وصياغة الأسئلة التحليلية...', 'info');
      setTimeout(() => {
        appState.batchIndex = 1;
        appState.generatedQuestions = synthesizePedagogicalQuestions(text, appState.batchIndex);
        if (appState.generatedQuestions.length === 0) return showToast('يرجى تحديد نوع واحد على الأقل من الأسئلة', 'error');

        el('bloomCountBadge').textContent = appState.generatedQuestions.length;
        el('bloomCountBadge').classList.remove('hidden');
        el('bloomBadgeNav').textContent = appState.generatedQuestions.length;
        el('bloomBadgeNav').classList.remove('hidden');

        toggleBloomSubMode('interactive');
        startConfetti();
        showToast(`تم توليد ${appState.generatedQuestions.length} سؤال تحليلي عالي المستوى!`, 'success');
      }, 400);
    };

    window.generateFreshBatch = () => {
      if (!appState.extractedRawText.trim()) return showToast('لا يوجد محتوى لتوليد دفعة جديدة منه', 'error');
      appState.batchIndex++;
      showToast(`جارٍ توليد دفعة جديدة تماماً (الدفعة ${appState.batchIndex})...`, 'info');
      setTimeout(() => {
        appState.generatedQuestions = synthesizePedagogicalQuestions(appState.extractedRawText, appState.batchIndex);
        renderQuestionsView();
        showToast('تم تحديث بنك الأسئلة بدفعة غير مكررة!', 'success');
      }, 350);
    };

        return showToast('يرجى توليد أسئلة بلوم أولاً ليتم تحويلها', 'error');

    };

    window.renderQuestionsView = () => {
      const container = el('questionsContainer');
      const emptyState = el('quizEmptyState');
      const filter = el('filterTypeSelect').value;

      if (!appState.generatedQuestions || appState.generatedQuestions.length === 0) {
        container.innerHTML = '';
        emptyState.classList.remove('hidden');
        return;
      }

      emptyState.classList.add('hidden');
      const filtered = filter === 'all' ? appState.generatedQuestions : appState.generatedQuestions.filter(q => q.typeKey === filter);

      if (filtered.length === 0) {
        container.innerHTML = `<div class="bg-white rounded-2xl p-8 text-center text-slate-500 border border-slate-200 font-bold text-xs sm:text-sm">لا توجد أسئلة تحت هذا التصنيف في الدفعة الحالية.</div>`;
        return;
      }

      container.innerHTML = filtered.map((q, index) => `
        <div class="bg-white rounded-3xl p-6 sm:p-8 border border-slate-200 shadow-sm transition hover:shadow-md relative overflow-hidden" id="q-card-${q.id}">
          <div class="flex flex-wrap items-center justify-between gap-2 mb-4 pb-3 border-b border-slate-100">
            <div class="flex items-center gap-2">
              <span class="w-7 h-7 rounded-xl bg-brand-50 text-brand-700 flex items-center justify-center text-xs font-black">${index + 1}</span>
              <span class="text-xs font-black text-slate-800">${q.typeName}</span>
            </div>
            <div class="flex items-center gap-1.5 px-3 py-1 rounded-full text-[11px] font-bold border ${q.bloomColor}">
              <i class="fa-solid ${q.bloomIcon}"></i>
              <span>مستوى بلوم: ${q.bloomText}</span>
            </div>
          </div>

          <div class="text-slate-900 font-bold text-sm sm:text-base mb-4 leading-relaxed bg-slate-50/70 p-4 rounded-2xl border border-slate-100">
            ${q.prompt}
          </div>

          <div class="mb-4">
            <label class="block text-xs font-bold text-slate-500 mb-1 flex items-center gap-1">
              <i class="fa-regular fa-pen-to-square text-indigo-500"></i>
              <span>مساحة التدوين وإجابة الطالب (تفاعلي):</span>
            </label>
            <textarea oninput="handleUserAnswerInput(${q.id}, this.value)" rows="2" class="w-full text-xs sm:text-sm p-3.5 rounded-xl border border-slate-200 focus:border-brand-500 focus:ring-2 focus:ring-brand-100 outline-none text-slate-800 bg-white" placeholder="اكتب تحليلك وإجابتك بأسلوبك الخاص قبل كشف النموذج الإرشادي...">${q.userAnswer || ''}</textarea>
          </div>

          <div class="flex flex-wrap items-center justify-between gap-3 pt-1">
            <button onclick="toggleQuestionReveal(${q.id})" class="px-4 py-2 rounded-xl text-xs font-black transition flex items-center gap-2 ${q.isRevealed ? 'bg-slate-100 hover:bg-slate-200 text-slate-700' : 'bg-brand-600 hover:bg-brand-700 text-white shadow-md shadow-brand-500/20'}">
              <i class="fa-solid ${q.isRevealed ? 'fa-eye-slash' : 'fa-lightbulb'}"></i>
              <span>${q.isRevealed ? 'إخفاء الإجابة النموذجية' : 'كشف الإجابة والتحليل التربوي'}</span>
            </button>
            <span class="text-[11px] text-slate-400 font-bold"><i class="fa-solid fa-graduation-cap text-indigo-400"></i> معايير نواتج التعلم</span>
          </div>

          <div id="answer-drawer-${q.id}" class="${q.isRevealed ? 'block' : 'hidden'} mt-4 pt-4 border-t-2 border-dashed border-indigo-100 bg-gradient-to-br from-indigo-50/40 via-white to-emerald-50/30 p-4 rounded-2xl fade-in text-xs sm:text-sm">
            <div class="flex items-center gap-1.5 text-emerald-800 font-black mb-1">
              <i class="fa-solid fa-circle-check text-emerald-600"></i>
              <span>الإجابة النموذجية المعتمدة:</span>
            </div>
            <p class="text-slate-800 font-semibold mb-2 leading-relaxed">${q.expectedAnswer}</p>
            ${q.correctionNote ? `<div class="text-xs text-slate-600 mb-2 bg-white/80 p-2.5 rounded-lg border border-slate-200/60"><span class="font-bold text-slate-800">ملاحظات التقييم:</span> ${q.correctionNote}</div>` : ''}
            <div class="text-[11px] text-indigo-900 bg-indigo-100/50 p-2.5 rounded-xl border border-indigo-200/60 leading-relaxed font-bold">
              <i class="fa-solid fa-chalkboard-user ml-1 text-indigo-700"></i>
              <span>توجيه المعلم الخبير:</span> ${q.rationale}
            </div>
          </div>
        </div>
      `).join('');
    };

    window.handleUserAnswerInput = (id, text) => {
      const q = appState.generatedQuestions.find(item => item.id === id);
      if (q) q.userAnswer = text;
    };

    window.toggleQuestionReveal = (id) => {
      const q = appState.generatedQuestions.find(item => item.id === id);
      if (!q) return;
      q.isRevealed = !q.isRevealed;
      renderQuestionsView();
    };

    window.revealAllAnswers = () => {
      if (!appState.generatedQuestions.length) return;
      const allRevealed = appState.generatedQuestions.every(q => q.isRevealed);
      appState.generatedQuestions.forEach(q => q.isRevealed = !allRevealed);
      renderQuestionsView();
      showToast(allRevealed ? 'تم إخفاء جميع الإجابات' : 'تم إظهار جميع الإجابات والتحليلات', 'info');
    };

      appState.includePrintAnswers = !appState.includePrintAnswers;
      el('printAnswersBtnText').textContent = appState.includePrintAnswers ? "إدراج نموذج الإجابة: نعم" : "إدراج نموذج الإجابة: لا (ورقة أسئلة فقط)";

        el('printAnswerKeySection').classList.remove('hidden');

        el('printAnswerKeySection').classList.add('hidden');
      }
    };

    function renderPrintView() {
      const printList = el('printQuestionsList');
      const printAnswers = el('printAnswersList');
      if (!appState.generatedQuestions.length) {
        printList.innerHTML = `<p class="text-center text-slate-400 py-8 text-sm font-bold">لا توجد أسئلة حالية لعرضها في ورقة الطباعة. يرجى توليد الأسئلة من تبويب بنك بلوم أولاً.</p>`;
        printAnswers.innerHTML = '';
        return;
      }

      const grouped = {};
      appState.generatedQuestions.forEach(q => {
        if (!grouped[q.typeName]) grouped[q.typeName] = [];
        grouped[q.typeName].push(q);
      });

      let questionHtml = '';
      let groupIndex = 1;
      for (const [groupName, qArr] of Object.entries(grouped)) {
        questionHtml += `
          <div class="mb-6">
            <h4 class="font-black text-slate-900 text-sm sm:text-base mb-3 pb-1 border-b border-slate-300 flex items-center justify-between">
              <span>السؤال رقم (${groupIndex}): ${groupName}</span>
              <span class="text-xs font-normal text-slate-500">درجة التقييم: (${qArr.length * 2} درجات)</span>
            </h4>
            <div class="space-y-4 pr-2">
              ${qArr.map((q, idx) => `
                <div class="text-xs sm:text-sm leading-relaxed">
                  <div class="font-bold text-slate-800 mb-1 flex items-start gap-2">
                    <span class="text-slate-500">${idx + 1}-</span>
                    <span>${q.prompt}</span>
                  </div>
                  <div class="h-10 border-b border-dotted border-slate-400 w-full mb-1"></div>
                </div>
              `).join('')}
            </div>
          </div>
        `;
        groupIndex++;
      }
      printList.innerHTML = questionHtml;

      printAnswers.innerHTML = appState.generatedQuestions.map((q, idx) => `
        <div class="p-3 rounded-lg bg-slate-50 border border-slate-200">
          <div class="flex items-center justify-between font-bold text-xs text-slate-700 mb-1">
            <span>س${idx + 1}: ${q.typeName}</span>
            <span class="text-[10px] bg-slate-200 px-2 py-0.5 rounded font-black">${q.bloomText}</span>
          </div>
          <p class="text-xs font-semibold text-slate-900 mb-1"><span class="text-emerald-700 font-bold">النموذج:</span> ${q.expectedAnswer}</p>
          <p class="text-[11px] text-slate-500 font-bold"><span class="font-bold">التوجيه:</span> ${q.rationale}</p>
        </div>
      `).join('');
    }

    // Confetti Animation Effect
    let confettiInterval;
    let particles = [];
    const canvas = el('confetti-canvas');
    const ctx = canvas.getContext('2d');
    const resizeCanvas = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; };
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    const startConfetti = () => {
      particles = [];
      const colors = ['#fde047', '#3b82f6', '#ec4899', '#10b981', '#a855f7', '#fb923c'];
      for (let i = 0; i < 150; i++) {
        particles.push({
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height - canvas.height,
          w: Math.random() * 10 + 6,
          h: Math.random() * 10 + 6,
          color: colors[Math.floor(Math.random() * colors.length)],
          speed: Math.random() * 4 + 3,
          angle: Math.random() * 360,
          spin: Math.random() * 0.2 - 0.1,
          shape: Math.random() > 0.5 ? 'rect' : 'circle'
        });
      }
      if (!confettiInterval) confettiInterval = requestAnimationFrame(updateConfetti);
      setTimeout(stopConfetti, 4000);
    };

    const stopConfetti = () => {
      cancelAnimationFrame(confettiInterval);
      confettiInterval = null;
      ctx.clearRect(0, 0, canvas.width, canvas.height);
    };

    const updateConfetti = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      particles.forEach((p) => {
        p.y += p.speed;
        p.angle += p.spin;
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.angle);
        ctx.fillStyle = p.color;
        if (p.shape === 'rect') ctx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
        else { ctx.beginPath(); ctx.arc(0, 0, p.w / 2, 0, Math.PI * 2); ctx.fill(); }
        ctx.restore();
        if (p.y > canvas.height) { p.y = -20; p.x = Math.random() * canvas.width; }
      });
      if (confettiInterval) confettiInterval = requestAnimationFrame(updateConfetti);
    };
